```bash
$0 #имя самого скрипта
$1 #первый аргумент
$2 #второй
$# #количество аргументов
$@ #все скопом

#Кавычки вокруг "$1" важны: без них пробелы в аргументе разъедутся на отдельные слова.

echo "$1" #добавляет \n в конец. 
printf '%s' "$1" #нет.

#Хороший тон: проверять $# и подсказывать usage, если аргументов нет: 
[ $# -lt 1 ] && { echo "usage: $0 <arg>"; exit 1; }
```
#### Пример качественного скрипта:

```bash
cat > script.sh << 'EOF'
#!/bin/sh

[ $# -lt 1 ] && { echo "usage: $0 <arg>"; exit 1; }

printf '%s' "$1" > test.txt
EOF

# [ $# -lt 1 ] Проверяет количество аргументов, -lt 1 - минимум 1
# && - если 1 команда успех (код 0), то выполняется 2:
# { echo "usage: $0 <arg>"; exit 1; } 
# печатается текст из echo и завешается с комдом ошибки 1 (stderr)
# printf '%s' - печать без перевода строки
```
