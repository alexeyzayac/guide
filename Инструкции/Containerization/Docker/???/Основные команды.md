```bash
#Узнать версию Docker
docker version

#Узнать все запущенные контейнеры
docker ps

#Узнать все запущенные и остановленне контейнеры
docker ps -a

#Узнать локальные образы docekr на компьютере
docker images

#скачает образ "hello-world" с последнем тэгом
docker pull hello-world

#скачает и запустит образ "hello-world" с последнем тэгом
docker run hello-world

#Удаление остановленного контейнера
docker rm <имя или id контейнера> 
```
